Send email from your app

This service sends your project's transactional email. You make one HTTPS call and the service validates, queues, sends, and tracks delivery. It also skips addresses that have bounced or complained. This guide covers Node.js, Python and Go.

Base URL https://email.example.com
Node.js 18 or later. No dependencies.

Before you start

Ask the service admin for two things:

  1. A project with a verified sending domain, for example billing.example.com. You can send from any address on that domain or its subdomains, and from nothing else.
  2. An API key for that project. It starts with es_live_ and is shown only once, so put it straight into your secret store.

Give your app both values as environment variables:

.env
EMAIL_SERVICE_URL=https://email.example.com
EMAIL_SERVICE_KEY=es_live_xxxxxxxxxxxxxxxxxxxxxxxx

Treat the key like a password. Use it only from server-side code, never from a browser or mobile app. Anyone holding it can send mail as your project.

Set up a client

The API is plain JSON over HTTPS, so you don't need an SDK. Copy this small helper into your project. It adds the key to every request and turns error responses into an exception you can inspect. Every example below uses it.

emailService.js
const BASE = process.env.EMAIL_SERVICE_URL.replace(/\/+$/, '');
const KEY = process.env.EMAIL_SERVICE_KEY;

export class EmailServiceError extends Error {
  constructor(status, body) {
    super(body?.error?.message ?? `HTTP ${status}`);
    this.status = status;
    this.code = body?.error?.code;
    this.details = body?.error?.details;
  }
}

export async function api(method, path, body) {
  const res = await fetch(BASE + path, {
    method,
    headers: {
      authorization: `Bearer ${KEY}`,
      ...(body !== undefined && { 'content-type': 'application/json' }),
    },
    body: body === undefined ? undefined : JSON.stringify(body),
    signal: AbortSignal.timeout(10_000),
  });
  if (res.status === 204) return null;
  const data = await res.json().catch(() => null);
  if (!res.ok) throw new EmailServiceError(res.status, data);
  return data;
}

email_service.py
import os

import requests

BASE = os.environ["EMAIL_SERVICE_URL"].rstrip("/")
KEY = os.environ["EMAIL_SERVICE_KEY"]

_session = requests.Session()
_session.headers["Authorization"] = f"Bearer {KEY}"


class EmailServiceError(Exception):
    def __init__(self, status, body):
        err = (body or {}).get("error") or {}
        super().__init__(err.get("message") or f"HTTP {status}")
        self.status = status
        self.code = err.get("code")
        self.details = err.get("details")


def api(method, path, body=None, params=None):
    res = _session.request(method, BASE + path, json=body, params=params, timeout=10)
    if res.status_code == 204:
        return None
    try:
        data = res.json()
    except ValueError:
        data = None
    if not res.ok:
        raise EmailServiceError(res.status_code, data)
    return data
emailservice/client.go
package emailservice

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"strings"
	"time"
)

type Client struct {
	BaseURL string
	APIKey  string
	HTTP    *http.Client
}

// New reads EMAIL_SERVICE_URL and EMAIL_SERVICE_KEY from the environment.
func New() *Client {
	return &Client{
		BaseURL: strings.TrimRight(os.Getenv("EMAIL_SERVICE_URL"), "/"),
		APIKey:  os.Getenv("EMAIL_SERVICE_KEY"),
		HTTP:    &http.Client{Timeout: 10 * time.Second},
	}
}

// APIError is returned for any 4xx or 5xx response.
type APIError struct {
	Status  int             `json:"-"`
	Code    string          `json:"code"`
	Message string          `json:"message"`
	Details json.RawMessage `json:"details,omitempty"`
}

func (e *APIError) Error() string {
	return fmt.Sprintf("email service: %d %s: %s", e.Status, e.Code, e.Message)
}

// Do sends a JSON request. in may be nil; out may be nil to discard the body.
func (c *Client) Do(ctx context.Context, method, path string, in, out any) error {
	var body io.Reader
	if in != nil {
		b, err := json.Marshal(in)
		if err != nil {
			return err
		}
		body = bytes.NewReader(b)
	}
	req, err := http.NewRequestWithContext(ctx, method, c.BaseURL+path, body)
	if err != nil {
		return err
	}
	req.Header.Set("Authorization", "Bearer "+c.APIKey)
	if in != nil {
		req.Header.Set("Content-Type", "application/json")
	}
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 400 {
		var e struct {
			Error APIError `json:"error"`
		}
		_ = json.NewDecoder(resp.Body).Decode(&e)
		e.Error.Status = resp.StatusCode
		if e.Error.Message == "" {
			e.Error.Message = http.StatusText(resp.StatusCode)
		}
		return &e.Error
	}
	if out == nil || resp.StatusCode == http.StatusNoContent {
		return nil
	}
	return json.NewDecoder(resp.Body).Decode(out)
}

Go 1.21 or later, standard library only. Next, add the request and response types. The omitempty tags matter: optional fields must be left out of the JSON, not sent as null.

emailservice/types.go
package emailservice

import (
	"context"
	"time"
)

type SendRequest struct {
	From           string            `json:"from"`
	To             []string          `json:"to"`
	Cc             []string          `json:"cc,omitempty"`
	Bcc            []string          `json:"bcc,omitempty"`
	ReplyTo        []string          `json:"reply_to,omitempty"`
	Subject        string            `json:"subject,omitempty"`
	HTML           string            `json:"html,omitempty"`
	Text           string            `json:"text,omitempty"`
	Template       string            `json:"template,omitempty"`
	TemplateData   map[string]any    `json:"template_data,omitempty"`
	Headers        map[string]string `json:"headers,omitempty"`
	Tags           map[string]string `json:"tags,omitempty"`
	IdempotencyKey string            `json:"idempotency_key,omitempty"`
}

type Email struct {
	ID                   string            `json:"id"`
	Status               string            `json:"status"`
	From                 string            `json:"from"`
	To                   []string          `json:"to"`
	SuppressedRecipients []string          `json:"suppressed_recipients"`
	Subject              string            `json:"subject"`
	Template             string            `json:"template"`
	Tags                 map[string]string `json:"tags"`
	Attempts             int               `json:"attempts"`
	LastError            string            `json:"last_error"`
	IdempotencyKey       string            `json:"idempotency_key"`
	CreatedAt            time.Time         `json:"created_at"`
	SentAt               *time.Time        `json:"sent_at"`
	IdempotentReplay     bool              `json:"idempotent_replay"`
}

func (c *Client) Send(ctx context.Context, msg SendRequest) (*Email, error) {
	var e Email
	if err := c.Do(ctx, "POST", "/v1/emails", msg, &e); err != nil {
		return nil, err
	}
	return &e, nil
}

func (c *Client) GetEmail(ctx context.Context, id string) (*Email, error) {
	var e Email
	if err := c.Do(ctx, "GET", "/v1/emails/"+id, nil, &e); err != nil {
		return nil, err
	}
	return &e, nil
}

Send an email

POST /v1/emails with a sender, at least one recipient, a subject and an HTML body, a text body, or both. Always send a text part as well when you can: some clients show only text, and spam filters score HTML-only mail worse.

  from: 'Billing <no-reply@billing.example.com>',

  subject: 'Your invoice #42',
  html: '<p>Hi Ada, your invoice for €120 is ready.</p>',
  text: 'Hi Ada, your invoice for €120 is ready.',
  tags: { invoice: '42' },

console.log(email.id, email.status); // "8f1c…", "queued"

Read the response

The call returns as soon as the email is stored and queued. Sending happens in the background, usually within seconds. A successful call has one of two status codes, and your code should accept both:

  • 202 Accepted: a new email was created.
  • 200 OK with "idempotent_replay": true: an earlier request already used this idempotency_key. You get the original email back, and nothing is sent again.

Suppressed recipients

The service keeps a suppression list that all projects share. An address goes on it after a hard bounce or a spam complaint. Suppressed addresses are removed from the email before it is queued, and they are listed in suppressed_recipients. If every to address is suppressed, the status is "suppressed" and nothing is sent. The call still returns success, so check for this case when it matters to you, for example in a password reset flow:

send.js
if (email.status === 'suppressed') {
  // Nothing was sent. Ask the user for another address.
} else if (email.suppressed_recipients.length) {
  console.warn('Skipped suppressed recipients:', email.suppressed_recipients);
}
send.py
if email["status"] == "suppressed":
    # Nothing was sent. Ask the user for another address.
    ...
elif email["suppressed_recipients"]:
    print("Skipped suppressed recipients:", email["suppressed_recipients"])
main.go
if email.Status == "suppressed" {
	// Nothing was sent. Ask the user for another address.
} else if len(email.SuppressedRecipients) > 0 {
	log.Printf("skipped suppressed recipients: %v", email.SuppressedRecipients)
}

Retries and idempotency

If a request times out, you can't tell whether the email was stored. Retrying without protection could send it twice. To prevent that, send an idempotency_key with every email. If the service sees a key again, it returns the email it already stored and sends nothing new.

The best key comes from the event that triggers the email, such as invoice-42-sent or password-reset-<token id>. A key like that also protects you when a job runs twice or your own process crashes and restarts. If there's no natural key, the helper below makes a random one for each email and reuses it on every retry of that email.

Retry network errors and 5xx responses. Don't retry 4xx responses, because the same request will fail the same way. Add this to the helper file, then use it for every send:

emailService.js
import { randomUUID } from 'node:crypto';

export async function sendEmail(message, { attempts = 4 } = {}) {
  const body = { idempotency_key: randomUUID(), ...message };
  for (let attempt = 1; ; attempt++) {
    try {
      return await api('POST', '/v1/emails', body);
    } catch (err) {
      const retryable = !(err instanceof EmailServiceError) || err.status >= 500;
      if (!retryable || attempt >= attempts) throw err;
      await new Promise((r) => setTimeout(r, 250 * 2 ** attempt));
    }
  }
}
send.js
import { sendEmail } from './emailService.js';

await sendEmail({
  from: 'no-reply@billing.example.com',
  to: ['ada@customer.com'],
  subject: 'Your invoice #42',
  text: 'Hi Ada, your invoice is ready.',
  idempotency_key: 'invoice-42-sent',
});
email_service.py
import time
import uuid


def send_email(message, attempts=4):
    body = {"idempotency_key": str(uuid.uuid4()), **message}
    for attempt in range(1, attempts + 1):
        try:
            return api("POST", "/v1/emails", body)
        except EmailServiceError as err:
            if err.status < 500 or attempt == attempts:
                raise
        except requests.RequestException:
            if attempt == attempts:
                raise
        time.sleep(0.25 * 2 ** attempt)

emailservice/retry.go
package emailservice

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"errors"
	"time"
)

// SendWithRetry retries network errors and 5xx responses. It reuses one
// idempotency key across attempts, so the email is sent at most once.
func (c *Client) SendWithRetry(ctx context.Context, msg SendRequest, attempts int) (*Email, error) {
	if msg.IdempotencyKey == "" {
		b := make([]byte, 16)
		if _, err := rand.Read(b); err != nil {
			return nil, err
		}
		msg.IdempotencyKey = hex.EncodeToString(b)
	}
	var lastErr error
	for attempt := 1; attempt <= attempts; attempt++ {
		email, err := c.Send(ctx, msg)
		if err == nil {
			return email, nil
		}
		var apiErr *APIError
		if errors.As(err, &apiErr) && apiErr.Status < 500 {
			return nil, err // a 4xx won't succeed on retry
		}
		lastErr = err
		if attempt < attempts {
			select {
			case <-time.After(250 * time.Millisecond << attempt):
			case <-ctx.Done():
				return nil, ctx.Err()
			}
		}
	}
	return nil, lastErr
}

Templates

You can store a template once and then send it by name with different data each time. Templates use Handlebars. In the HTML body, {{name}} is HTML-escaped. Use {{{name}}} only for HTML you trust. Templates belong to your project, and PUT creates a template or replaces the existing one.

templates.js
import { api, sendEmail } from './emailService.js';

// Create or update (run once, e.g. from a deploy script)
await api('PUT', '/v1/templates/invoice', {
  subject: 'Your invoice #{{number}}',
  html: '<p>Hi {{name}}, invoice {{number}} for {{amount}} is ready.</p>',
  text: 'Hi {{name}}, invoice {{number}} for {{amount}} is ready.',
});

// Preview without sending
const preview = await api('POST', '/v1/templates/invoice/render', {
  data: { name: 'Ada', number: 42, amount: '€120' },
});
console.log(preview.subject); // "Your invoice #42"

// Send it
await sendEmail({
  from: 'no-reply@billing.example.com',
  to: ['ada@customer.com'],
  template: 'invoice',
  template_data: { name: 'Ada', number: 42, amount: '€120' },
  idempotency_key: 'invoice-42-sent',
});

The template is rendered at the moment you send. Editing it later doesn't change emails that are already queued. If the template doesn't exist, the send fails with 404 not_found. A template with a syntax error is refused when you save it, with 400 template_syntax. You can also call GET /v1/templates to list your templates, and GET or DELETE /v1/templates/:name to read or remove one.

Track delivery

Each email moves through these statuses:

queued → sending → sent → delivered | bounced | complained ↘ failed rejected for sending, or every retry failed suppressed every "to" address was suppressed; nothing sent

sent means the email has left the service on its way to the recipient. delivered, bounced and complained come later, when the recipient's mail server reports back. That can take seconds or, for a soft bounce, much longer. Your project doesn't get a callback when the status changes. To find out, fetch the email by its id:

status.js
const email = await api('GET', `/v1/emails/${id}`);
console.log(email.status, email.attempts, email.last_error);

// Delivery events: Delivery, Bounce, Complaint, …
const { data: events } = await api('GET', `/v1/emails/${id}/events`);
for (const e of events) console.log(e.created_at, e.event_type);
status.py
email = api("GET", f"/v1/emails/{email_id}")
print(email["status"], email["attempts"], email["last_error"])

# Delivery events: Delivery, Bounce, Complaint, …
for e in api("GET", f"/v1/emails/{email_id}/events")["data"]:
    print(e["created_at"], e["event_type"])
main.go
email, err := client.GetEmail(ctx, id)
if err != nil {
	log.Fatal(err)
}
fmt.Println(email.Status, email.Attempts, email.LastError)

// Delivery events: Delivery, Bounce, Complaint, …
var events struct {
	Data []struct {
		EventType string          `json:"event_type"`
		Payload   json.RawMessage `json:"payload"`
		CreatedAt time.Time       `json:"created_at"`
	} `json:"data"`
}
if err := client.Do(ctx, "GET", "/v1/emails/"+id+"/events", nil, &events); err != nil {
	log.Fatal(err)
}
for _, e := range events.Data {
	fmt.Println(e.CreatedAt, e.EventType)
}

The single-email response also includes the rendered html, text, headers and template_data, so you can see exactly what was sent.

List your emails

GET /v1/emails returns your project's emails, newest first, without their bodies. You can filter by status and to, and set limit from 1 to 200 (default 50). To get the next page, pass the response's next_before back as before. When there are no more pages, next_before is null.

list.js
// Every bounced email, page by page
let before = null;
do {
  const params = new URLSearchParams({ status: 'bounced', limit: '100' });
  if (before) params.set('before', before);
  const page = await api('GET', `/v1/emails?${params}`);
  for (const m of page.data) console.log(m.created_at, m.to.join(', '), m.subject);
  before = page.next_before;
} while (before);
list.py
# Every bounced email, page by page
params = {"status": "bounced", "limit": 100}
while True:
    page = api("GET", "/v1/emails", params=params)
    for m in page["data"]:
        print(m["created_at"], ", ".join(m["to"]), m["subject"])
    if not page["next_before"]:
        break
    params["before"] = page["next_before"]
main.go
// Every bounced email, page by page
params := url.Values{"status": {"bounced"}, "limit": {"100"}}
for {
	var page struct {
		Data       []emailservice.Email `json:"data"`
		NextBefore *string              `json:"next_before"`
	}
	if err := client.Do(ctx, "GET", "/v1/emails?"+params.Encode(), nil, &page); err != nil {
		log.Fatal(err)
	}
	for _, m := range page.Data {
		fmt.Println(m.CreatedAt, strings.Join(m.To, ", "), m.Subject)
	}
	if page.NextBefore == nil {
		break
	}
	params.Set("before", *page.NextBefore)
}

Suppressions

You don't need to call these endpoints to send safely, because suppressed addresses are skipped automatically. They are useful when you want to check an address before sign-up, or to stop mail to someone who asked you to.

  • GET /v1/suppressions/:email returns {"email": "…", "suppressed": true|false}.
  • POST /v1/suppressions with {"email": "…", "note": "…"} adds a manual suppression and returns 201. The list is shared, so this stops mail to that address from every project.
  • DELETE /v1/suppressions/:email returns 204. You can only remove manual suppressions that your project added. For bounces, complaints and other projects' entries you get 403 suppression_not_removable. Ask an admin to remove those.

The address goes in the URL path, so URL-encode it. Characters that email addresses may contain, such as /, #, ? and %, would otherwise break the URL.

suppressions.js
const path = `/v1/suppressions/${encodeURIComponent(address)}`;

const { suppressed } = await api('GET', path);

await api('POST', '/v1/suppressions', { email: address, note: 'Unsubscribed in settings' });

await api('DELETE', path);
suppressions.py
from urllib.parse import quote

path = f"/v1/suppressions/{quote(address, safe='')}"

suppressed = api("GET", path)["suppressed"]

api("POST", "/v1/suppressions", {"email": address, "note": "Unsubscribed in settings"})

api("DELETE", path)
main.go
path := "/v1/suppressions/" + url.PathEscape(address)

var check struct {
	Suppressed bool `json:"suppressed"`
}
if err := client.Do(ctx, "GET", path, nil, &check); err != nil {
	log.Fatal(err)
}

body := map[string]string{"email": address, "note": "Unsubscribed in settings"}
if err := client.Do(ctx, "POST", "/v1/suppressions", body, nil); err != nil {
	log.Fatal(err)
}

if err := client.Do(ctx, "DELETE", path, nil, nil); err != nil {
	log.Fatal(err) // 403 suppression_not_removable if your project didn't add it
}

Errors

Every error response has the same shape. Branch on code, not on message, because messages may change.

403 response
{

StatusCodeMeaning
400validation_errorThe body doesn't match the schema: a wrong type, a missing field or a value that is too long. details lists each problem.
400invalid_fromfrom isn't a valid address.
400invalid_recipientOne or more of to, cc, bcc or reply_to isn't a valid address. details.addresses lists them.
400missing_recipientNo usable to address.
400too_many_recipientsMore than 50 addresses across to, cc and bcc.
400missing_subject, missing_bodyNo subject, or no html and no text, and no template.
400reserved_header, invalid_headerA header you can't set, or a malformed header name or value.
400empty_template, template_syntaxThe template renders to nothing, or doesn't compile when you save it.
401unauthorizedThe API key is missing, malformed or revoked.
403sender_domain_not_allowedThe from domain isn't one of your project's domains.
403suppression_not_removableYou tried to remove a suppression your project didn't add.
404not_foundThe email, template or suppression doesn't exist, or belongs to another project.
5xxinternal_errorA problem on the service's side. Retry with the same idempotency_key.

API reference

Every endpoint needs Authorization: Bearer es_live_…. Request and response bodies are JSON.

MethodPathPurpose
POST/v1/emailsQueue an email. Returns 202, or 200 on an idempotent replay.
GET/v1/emailsList emails. Query: limit, before, status, to.
GET/v1/emails/:idOne email, with its body, attempts and last error.
GET/v1/emails/:id/eventsDelivery events for the email: delivery, bounce, complaint.
GET/v1/templatesList templates.
GET PUT DELETE/v1/templates/:nameRead, create or replace, or delete a template.
POST/v1/templates/:name/renderPreview a template with {"data": {…}}.
GET/v1/suppressions/:emailCheck whether an address is suppressed.
POST/v1/suppressionsAdd a manual suppression.
DELETE/v1/suppressions/:emailRemove a manual suppression that your project added.
